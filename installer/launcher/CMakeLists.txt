cmake_minimum_required(VERSION 3.20)
project(StarEffectsLabLauncher C)

set(PYTHON_HOME "" CACHE PATH "Python install prefix (holds include/ and lib/ (macOS) or libs/ (Windows))")
if(NOT PYTHON_HOME)
  message(FATAL_ERROR "Pass -DPYTHON_HOME=<python prefix>")
endif()

if(WIN32)
  file(GLOB _lib "${PYTHON_HOME}/libs/python3[0-9]*.lib")
  list(FILTER _lib EXCLUDE REGEX "_d\\.lib$")
  list(GET _lib 0 PY_LIB)
  get_filename_component(_name "${PY_LIB}" NAME_WE)
  set(PY_INCLUDE "${PYTHON_HOME}/include")
  set(ICON_FILE "" CACHE FILEPATH "application.ico embedded in the exe")
  if(NOT ICON_FILE)
    message(FATAL_ERROR "Pass -DICON_FILE=<application.ico>")
  endif()
  file(TO_CMAKE_PATH "${ICON_FILE}" ICON)
  configure_file(launcher.rc.in "${CMAKE_BINARY_DIR}/launcher.rc" @ONLY)
  add_executable(launcher WIN32 launcher.c "${CMAKE_BINARY_DIR}/launcher.rc")
  # Static CRT: no VC++ redistributable needed for the launcher itself.
  set_target_properties(launcher PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded")
  target_compile_definitions(launcher PRIVATE "PY_DLL_NAME=L\"${_name}.dll\"")
  # python3xx.dll lives in <root>\python, so load it lazily after launcher.c pre-loads it from there.
  target_link_libraries(launcher PRIVATE "${PY_LIB}" delayimp)
  target_link_options(launcher PRIVATE "/DELAYLOAD:${_name}.dll")
  target_link_libraries(launcher PRIVATE shell32 user32)
else()
  file(GLOB _inc "${PYTHON_HOME}/include/python3.*")
  file(GLOB _lib "${PYTHON_HOME}/lib/libpython3.*.dylib")
  list(GET _inc 0 PY_INCLUDE)
  list(GET _lib 0 PY_LIB)
  add_executable(launcher launcher.c)
  target_link_libraries(launcher PRIVATE "${PY_LIB}")
endif()

set_target_properties(launcher PROPERTIES OUTPUT_NAME "Star Effects Lab")
target_include_directories(launcher PRIVATE "${PY_INCLUDE}")
if(APPLE)
  set_target_properties(launcher PROPERTIES
    BUILD_WITH_INSTALL_RPATH ON
    INSTALL_RPATH "@executable_path/../Resources/python/lib")
endif()
